      set -euo pipefail
      
      if [ -x /opt/mssql-tools18/bin/sqlcmd ]; then
        SQLCMD="/opt/mssql-tools18/bin/sqlcmd"
        SQLCMD_FLAGS="-C"
      elif [ -x /opt/mssql-tools/bin/sqlcmd ]; then
        SQLCMD="/opt/mssql-tools/bin/sqlcmd"
        SQLCMD_FLAGS=""
      else
        echo "sqlcmd introuvable dans l'image SQL."
        exit 1
      fi
      
      TS="$(date +%Y%m%d-%H%M%S)"
      
      echo "[backup] Waiting for SQL Server..."
      for i in {1..60}; do
        if ${SQLCMD} ${SQLCMD_FLAGS} -S sql-data -U sa -P "${SA_PASSWORD}" -b -Q "SELECT 1" >/dev/null 2>&1; then
          echo "[backup] SQL Server is ready."
          break
        fi
      
        if [ "$i" -eq 60 ]; then
          echo "[backup] SQL Server not ready after timeout."
          exit 1
        fi
      
        sleep 5
      done
      
      backup_db() {
        DB_NAME="$1"
        BACKUP_FILE="$2"
      
        echo "[backup] Backing up database ${DB_NAME} to ${BACKUP_FILE}"
      
        ${SQLCMD} ${SQLCMD_FLAGS} \
          -S sql-data \
          -U sa \
          -P "${SA_PASSWORD}" \
          -b \
          -Q "BACKUP DATABASE [${DB_NAME}] TO DISK = N'${BACKUP_FILE}' WITH INIT, COMPRESSION, CHECKSUM, STATS = 10"
      
        echo "[backup] Backup completed for ${DB_NAME}"
      }
      backup_db "dotnetgigs.applicants" "${BACKUP_DIR}/dotnetgigs-applicants-${TS}.bak"
      backup_db "dotnetgigs.jobs" "${BACKUP_DIR}/dotnetgigs-jobs-${TS}.bak"
      
      echo "[backup] SQL backup job completed successfully."
